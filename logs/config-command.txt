import glob, os, sys
import xml.etree.ElementTree as ET

FAILED = ('failed', 'undefined')

def load(path, into):
    root = ET.parse(path).getroot()
    for case in root.iter('testcase'):
        key = case.get('file') + ':' + case.get('line')
        failed = case.get('status') in FAILED or case.find('failure') is not None
        into[(key, case.get('name') or '')] = {'suite': root.get('name'), 'name': case.get('name'), 'key': key, 'failed': failed}

def read_run(directory):
    scenarios, flaky, rerun = {}, set(), {}
    for path in sorted(glob.glob(os.path.join(directory, '*.xml'))):
        load(path, scenarios)
    for path in sorted(glob.glob(os.path.join(directory, 'rerun', '*.xml'))):
        load(path, rerun)
    # The rerun has the final say: failing and then passing is flaky, not failing.
    for key, scenario in rerun.items():
        if key in scenarios and scenarios[key]['failed'] and not scenario['failed']:
            flaky.add(key)
        scenarios[key] = scenario
    return scenarios, flaky

def failing(scenarios):
    return {key for key, scenario in scenarios.items() if scenario['failed']}

candidate, flaky = read_run(sys.argv[1])
baseline, _ = read_run(sys.argv[2])

if not candidate:
    print('::error::The run against the build being tested produced no results.')
    sys.exit(1)

broken = sorted(failing(candidate) - failing(baseline))
known = sorted(failing(candidate) & failing(baseline))

lines = []

    lines += ['### %d scenario(s) fail only with this build' % len(broken), '', '| Scenario | Location |', '| --- | --- |']
    lines += ['| %s | `%s` |' % (candidate[key]['name'], candidate[key]['key']) for key in broken]
    for key in broken:
        print('::error title=Fails only with this build::%s (%s)' % (candidate[key]['name'], candidate[key]['key']))
else:
    lines.append('### No new failures')
if known:
    lines += ['', '%d scenario(s) also fail against the baseline build, so this build did not cause them:' % len(known), '']
    lines += ['- %s (`%s`)' % (candidate[key]['name'], candidate[key]['key']) for key in known]
if flaky:
    lines += ['', '%d scenario(s) failed and then passed on the rerun, and were treated as flaky.' % len(flaky)]

summary = '\n'.join(lines) + '\n'
print(summary)
if os.environ.get('GITHUB_STEP_SUMMARY'):
